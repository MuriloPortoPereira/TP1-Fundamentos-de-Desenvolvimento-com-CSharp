//     Console.WriteLine("Opcao invalida!");
//     continue;

- if verifica se a condicao é verdadeira
- ! inverte o resultado do TryParse
- TryParse tenta converter o que foi digitado em numero inteiro
- out opcao guarda o numero convertido na variavel opcao

if (!int.TryParse(Console.ReadLine(), out opcao))
{
    // Se a conversao falhar, mostra a mensagem
    Console.WriteLine("Opcao invalida!");

    // Pula o restante do codigo e volta para o inicio do loop
    continue;
}

Então fica assim. (Entrada que vai ser analisada, Oque deve ser retornado / saida)
Ai fica !tipo.TryParse

O TryParse ve se e verdadeiro padrao, mas com ! ele fica invertido. Ai se for int, ele continua. Caso nao seja ele continua o programa (Fluxo do Codigo). Mas ele acaba ainda
Dando Continue (Ele nao quebra, apenas retorna a saida opcao "Recomeca")

{
case 1:
	{
	
	}
case 2:
	{

	}
default:
	{
	// default executa se nenhuma opcao corresponder.
	// Semelhante ao if e Else.
	}
}
