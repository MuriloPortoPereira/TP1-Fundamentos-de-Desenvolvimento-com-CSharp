// Sobre o Try Parse:
//
// Try = Tentar
// Parse = Converter ou interpretar
//
// Então, int.TryParse() pode ser entendido como "Tentar converter para número inteiro".
// 
// Já !int.TryParse() seria "Não conseguiu converter para número inteiro?"

// TRYPARSE - CONVERSAO DE TEXTO PARA NUMERO
//
// O TryParse tenta transformar um texto em numero.
// Por exemplo, "90" é um texto e 90 é um numero inteiro.
//
// string entrada = "90";
//
// bool conseguiu = int.TryParse(entrada, out int minutos);
//
// int       -> Tipo que queremos converter (numero inteiro).
// TryParse  -> Tenta fazer a conversao.
// entrada   -> Texto que queremos converter.
// out       -> Indica onde guardar o resultado convertido.
// minutos   -> Variavel que recebe o numero.
// conseguiu -> Recebe true ou false.
//
// Se entrada = "90":
// Retorna true e guarda 90 na variavel minutos.
//
// Se entrada = "abc":
// Retorna false e a variavel minutos recebe 0.
//
// RESUMINDO:
// TryParse tenta converter e responde se conseguiu.
// out guarda o valor convertido.
// O retorno true ou false indica se deu certo ou nao.
